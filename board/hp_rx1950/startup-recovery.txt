# One-shot HaRET configuration used by the authenticated cable updater.
set RAMADDR 0x30000000
set RAMSIZE 32*1024*1024
set MTYPE 952
set KERNEL_OFFSET 0x1000000
set KERNEL zImage.ota
set FBDURINGBOOT 0
set KERNELCRC 1
# zImage.ota is the dedicated recovery kernel with /init embedded.  Do not use
# HaRET INITRD here: its ATAG_INITRD2 is lost on this machine.
set CMDLINE "console=tty0 loglevel=4 consoleblank=0"
# Recovery is launched by the same early Windows Mobile autostart path.
sleep 10000
bootlinux
