# HaRET startup script for the SD card root directory.
# RX1950 uses a 32 MiB SDRAM window at 0x30000000.
set RAMADDR 0x30000000
set RAMSIZE 32*1024*1024
set MTYPE 952
set KERNEL_OFFSET 0x500000
set KERNEL zImage
set FBDURINGBOOT 0
set KERNELCRC 1
set CMDLINE "root=/dev/mmcblk0p2 rootwait rw console=ttySAC0,115200n8 console=tty0"
bootlinux
